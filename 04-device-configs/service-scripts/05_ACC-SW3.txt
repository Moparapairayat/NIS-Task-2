configure terminal
!
! ---- User ports: PortFast, BPDU Guard and port security (Section 25) ----
interface range fastEthernet0/1 - 4
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation shutdown
 exit
!
! ---- DHCP snooping: only the uplink may answer DHCP requests ----
ip dhcp snooping
ip dhcp snooping vlan 30
no ip dhcp snooping information option
interface gigabitEthernet0/1
 ip dhcp snooping trust
 exit
!
! ---- Secure remote management: SSH v2 from the Admin VLAN only (Section 25) ----
ip domain-name college.local
username admin privilege 15 secret Adm1n@2026
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
access-list 10 remark ADMIN VLAN ONLY FOR MANAGEMENT
access-list 10 permit 192.168.10.0 0.0.0.255
line vty 0 15
 access-class 10 in
 login local
 transport input ssh
 exec-timeout 10 0
 exit
!
! ---- Save ----
end
write memory
!
