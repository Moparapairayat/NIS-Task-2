! ---- Basic device setup (Section 18) ----
enable
configure terminal
hostname ISP-R
no ip domain-lookup
enable secret Netr0n@2026
service password-encryption
banner motd # Authorised access only. Unauthorised access is prohibited. #
line console 0
 password Con@2026
 login
 exec-timeout 10 0
 logging synchronous
 exit
!
! ---- Link to the campus edge router ----
interface gigabitEthernet0/0
 description LINK TO NETRON R1-EDGE
 ip address 203.0.113.1 255.255.255.252
 no shutdown
 exit
!
! ---- Simulated internet network ----
interface gigabitEthernet0/1
 description SIMULATED INTERNET
 ip address 198.51.100.1 255.255.255.0
 no shutdown
 exit
!
! ---- Save ----
end
write memory
!
! ---- Verification ----
! show ip interface brief
! ping 203.0.113.2
! ping 198.51.100.10
