! ---- Basic device setup (Section 16) ----
enable
configure terminal
hostname ACC-SW1
no ip domain-lookup
enable secret Netr0n@2026
service password-encryption
banner motd # Authorised access only. Unauthorised access is prohibited. #
line console 0
 password Con@2026
 login
 exec-timeout 10 0
 logging synchronous
 exit
!
! ---- VLANs (Table 3) ----
vtp mode transparent
vlan 10
 name ADMIN
 exit
vlan 99
 name MANAGEMENT
 exit
vlan 998
 name BLACKHOLE
 exit
vlan 999
 name NATIVE
 exit
!
! ---- Access ports for connected devices (Table 5) ----
interface range fastEthernet0/1 - 3
 description ADMIN PCs
 switchport mode access
 switchport access vlan 10
 no shutdown
 exit
!
! ---- Unused ports: VLAN 998 and shut down ----
interface range fastEthernet0/4 - 24
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
 exit
interface gigabitEthernet0/2
 description UNUSED
 switchport mode access
 switchport access vlan 998
 shutdown
 exit
!
! ---- Uplink trunk to the core ----
interface gigabitEthernet0/1
 description UPLINK TO CORE-SW1 Fa0/1
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,99,999
 switchport nonegotiate
 no shutdown
 exit
!
! ---- Management address (VLAN 99) ----
interface vlan 99
 description MANAGEMENT
 ip address 192.168.99.11 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.99.1
!

end
